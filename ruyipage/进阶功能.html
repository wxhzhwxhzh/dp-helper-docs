    print(cookie.name, cookie.value)

# 简单属性 / 浏览器级

page.browser.cookies(all_info=False)  # 浏览器级 cookies

page.delete_cookies()               # 不带参数 = 清空全部

12. 网络拦截与抓包​

参考示例 11_network_intercept.py、18_advanced_network.py。page.intercept 是核心网络拦截入口。

# 1) 拦截并 Mock 响应
def mock_handler(req):
    if "/api/data" in req.url:
        req.mock(
            '{"status":"ok","data":{"message":"mocked-by-interceptor"}}',
            status_code=200,
            headers={"content-type": "application/json",
                     "access-control-allow-origin": "*"},   # 跨域需带 CORS 头
        )
    else:
        req.continue_request()       # 不处理的请求必须显式继续

page.intercept.start(mock_handler, phases=["beforeRequestSent"])

# ... 触发请求 ...

page.intercept.stop()

# 2) 阻止请求
def fail_handler(req):
    if "/api/data" in req.url:
        req.fail()
    else:
        req.continue_request()

# 3) 修改请求头并继续（headers 支持 dict）
req.continue_request(headers={"X-Ruyi-Demo": "yes", "User-Agent": "RuyiPage/1.0"})

# 4) 队列模式：handler=None，用 wait() 手动取出
page.intercept.start(handler=None, phases=["beforeRequestSent"])
req = page.intercept.wait(timeout=5)
req.continue_request()

# 5) 响应阶段读取并修改
def response_handler(req):
    if "/api/data" in req.url:
        print(req.response_status)
        print(req.response_headers)
        req.continue_response(status_code=299, reason_phrase="RuyiModified")
    else:
        req.continue_response()

page.intercept.start_responses(response_handler)

# 6) 直接读取请求体
body = req.body          # 请求体
req.request_id           # 请求 ID

# 7) collect_response=True 时一步读取响应体
resp_body = req.response_body

常用拦截方法：

  • page.intercept.start(handler, phases=[...]) — 通用入口
  • page.intercept.start_requests(handler) — 请求阶段快捷方式
  • page.intercept.start_responses(handler) — 响应阶段快捷方式
  • page.intercept.wait(timeout) — 队列模式等待
  • page.intercept.stop() — 停止

22. 事件系统​

ruyipage 基于 BiDi 订阅了大量事件，覆盖导航、网络、脚本、日志、上下文、下载、用户提示等。参考示例 24_navigation_events.py、30_browsing_context_events.py、31_network_events.py、32_script_events.py、33_log_input_events.py。

通用事件订阅（page.events）：

# 订阅多个事件
page.events.start([
    "network.beforeRequestSent",
    "network.responseStarted",
    "network.responseCompleted",
    "network.fetchError",
    "network.authRequired",
], contexts=[page.tab_id])

page.events.clear()                    # 清空缓存
evt = page.events.wait("network.beforeRequestSent", timeout=5)   # 等待某事件
page.events.stop()                     # 停止

专用订阅器（更便捷）：

# 导航事件
page.navigation.start([...])
page.navigation.wait("browsingContext.domContentLoaded", timeout=3)
page.navigation.wait_for_load(timeout=3)
page.navigation.wait_for_fragment("a", timeout=3)

# realm 事件（脚本执行环境）
page.realms.start()
page.realms.list()
page.realms.stop()

# 日志 / 控制台事件
page.console.start()
page.console.wait(timeout=3)
page.console.entries

常见事件名：

  • 导航：navigationStarted、fragmentNavigated、historyUpdated、domContentLoaded、load、navigationCommitted、navigationFailed、navigationAborted
  • 网络：beforeRequestSent、responseStarted、responseCompleted、fetchError、authRequired
  • 脚本：realmCreated、realmDestroyed、message
  • 上下文：contextCreated、contextDestroyed、userPromptOpened、userPromptClosed
  • 下载：downloadWillBegin、downloadEnd
  • 输入：fileDialogOpened
  • 日志：log.entryAdded

23. 网络数据采集​

参考示例 28_network_data_collector.py、40_scraper_packet_capture.py、54_bing_passive_capture.py、53_duckai_eventstream_capture.py。

方式一：DataCollector + listen（手动编排）

from ruyipage import DataCollector, FirefoxPage, InterceptedRequest

collector = page.network.add_data_collector(data_types=["response"])
page.network.set_extra_headers({"X-Test-Collector": "yes"})
page.network.set_cache_behavior("bypass")

page.intercept.start_requests()
page.listen.start("/api/collector")

# ... 触发请求 ...

req = page.intercept.wait(timeout=8)          # 捕获请求
req.continue_request()

packet = page.listen.wait(timeout=8)          # 等待响应完成事件

data = collector.get(req.request_id, data_type="response")   # 按 request_id 取响应体
data.has_data
data.bytes
collector.disown(req.request_id, data_type="response")       # 释放
collector.remove()                             # 移除收集器

方式二：page.capture（被动抓包，最简单）

# 必须先 start，再打开页面（历史请求无法补抓）
page.capture.start("bing.com", method="GET", collect_bodies=True)
page.get(search_url)

# 等待抓包（count=1 返回单个，count>1 返回列表）
result = page.capture.wait(timeout=15, count=5)

packet.request_headers       # 请求头（dict，key 小写）
packet.request_body          # 请求体
packet.response_status       # 响应状态
packet.response_headers      # 响应头
packet.response_body         # 响应体（自动等待解码）
packet.is_failed

# 历史包
page.capture.steps
page.capture.stop()

方式三：拦截 + collect_response 一步到位（参考 40_scraper_packet_capture.py）

# collect_response=True 自动管理 DataCollector
page.intercept.start(handler=None, phases=["beforeRequestSent"], collect_response=True)
req = page.intercept.wait(timeout=8)
req.continue_request()
resp_body = req.response_body     # 无需手动编排，直接读响应体

25. 代理配置​

参考示例 38_proxy_auth_ipinfo.py、51_socks5_auth_proxy.py、52_per_tab_socks5_proxy_browserscan.py 及 http_socks5_examples/。

HTTP 代理（fpfile 认证）：

opts = FirefoxOptions()
opts.set_browser_path(BROWSER_PATH)
opts.set_proxy("http://host:port")
opts.set_fpfile(FPFILE_PATH)    # 内核自动读取 httpauth.username/password
page = FirefoxPage(opts)

fpfile 内容：

httpauth.host:gate.example.com
httpauth.port:1288
httpauth.username:<user>
httpauth.password:<pass>

SOCKS5 密码代理：

page = launch(browser_path=..., user_dir=profile_dir, fpfile=fpfile_path)

fpfile 内容（= 格式）：

socksauth.host=host
socksauth.port=1000
socksauth.username=<user>
socksauth.password=<pass>

按标签页分配不同代理（container tabs）：

opts = FirefoxOptions()
opts.set_browser_path(browser_path)
opts.set_per_tab_proxies(proxies, exhausted="wrap")   # 代理池
page = FirefoxPage(opts)

tabs = page.new_container_tabs(count=len(proxies), url=TARGET_URL)
# 每个 container tab 自动使用不同的代理

说明：按标签页代理需要支持 proxy.rotate.* 的 ruyi 定制 Firefox 指纹内核。

XPath Picker 是辅助定位复杂节点的工具，启动时打开，可在浏览器中手动点选并校验 XPath。

page = launch(
    headless=False,
    xpath_picker=True,          # 开启 XPath picker
    window_size=(1600, 1100),
)

典型用途（示例 44）：用 xpath 定位商品列表、翻页按钮并多页采集。

ITEMS_XPATH = "xpath://section[1]/ul[1]/li"
NEXT_PAGE_XPATH = "xpath:/html[1]/body[1]/div[1]/..."

items = page.eles(ITEMS_XPATH, timeout=2)
for item in items:
    print(item.text)

next_btn = page.ele(NEXT_PAGE_XPATH, timeout=5)
next_btn.click_self()

29. 鼠标行为可视化​

参考示例 42_2_action_visual_showcase.py、49_human_click_visual_grid.py、50_human_click_visual_points.py。

开启 action_visual=True 后，浏览器内会实时显示鼠标轨迹、点击位置扩散动画、目标元素高亮框、当前坐标等，非常适合调试拟人操作。

page = launch(
    action_visual=True,          # 开启可视化
    headless=False,
    port=9334,
    window_size=(1400, 900),
)

# 各种移动轨迹都能可视化
page.actions.move_to((200, 200)).perform()
page.actions.human_move((800, 200)).perform()
page.actions.move(100, 0).perform()

30. 拟人化行为​

参考示例 46_human_behavior_showcase.py。ruyipage 内置两种拟人鼠标轨迹算法：Bezier 和 WindMouse，配合 action_visual 可直观观察。

opts = FirefoxOptions()
opts.enable_action_visual(True)
opts.set_human_algorithm("bezier")          # 默认算法
opts.set_window_size(1440, 980)
page = FirefoxPage(opts)

# 按算法与风格移动 + 点击 + 输入
page.actions.human_move(page.ele("#target-a"),
                        algorithm="bezier", style="line_then_arc").human_click().perform()
page.actions.human_move(page.ele("#target-b"),
                        algorithm="windmouse").human_click().perform()
page.actions.human_move(page.ele("#search-input"),
                        algorithm="bezier", style="arc").human_click().human_type("demo").perform()

拟人动作方法：

  • human_move(target, algorithm=..., style=...) — 拟人移动
  • human_click() — 拟人点击
  • human_type(text) — 拟人输入

31. 智能指纹​

参考示例 48_smart_fingerprint.py。opts.smart_fingerprint() 一键生成指纹与 fpfile，覆盖代理出口 IP、国家 / 时区 / 经纬度、语言、22 套 Windows 真机硬件特征、UA、Canvas/Audio 种子等。

from ruyipage import FirefoxOptions, FirefoxPage

opts = FirefoxOptions()
opts.set_browser_path(BROWSER_PATH)
opts.set_port(9222)

ctx = opts.smart_fingerprint(
    proxy_host="127.0.0.1", proxy_port=7890,
    proxy_user=None, proxy_pwd=None,
    require_country=None,          # 出口 IP 国家校验（None=不校验）
    logger=print,
)

print(ctx.summary())
page = FirefoxPage(opts)
ctx.apply_emulation(page, logger=print)   # 叠加 screen/geolocation/locale/timezone/headers
page.get("https://www.google.com")

异常处理：CountryMismatchError（国家不匹配）、GeoError（geo 探测全部失败）。

32. 调试与诊断​

参考示例 47_trace_and_diagnostics.py。提供 Trace 时间线 和 失败快照（Failure Snapshot） 两套诊断能力。

三种启用方式：

# 方式 A：launch 参数（最简单）
page = launch(trace=True, failure_snapshot=True, snapshot_dir="./diag_output_47")

# 方式 B：FirefoxOptions 链式
opts = FirefoxOptions()
opts.enable_trace(True)
opts.enable_failure_snapshot()
opts.set_snapshot_dir("./diag")
page = FirefoxPage(opts)

# 方式 C：Settings 全局开关
Settings.trace_enabled = True
Settings.trace_max_entries = 2000
Settings.failure_snapshot_enabled = True
Settings.raise_when_ele_not_found = True   # 元素找不到时抛异常

Trace 时间线 API：

page.trace.enabled                # 是否启用
page.trace.latest(5)              # 最近 N 条 TraceEntry
page.trace.recent_requests(10)    # 最近 N 条网络请求
page.trace.summary(10)            # 人类可读摘要
page.trace.dump_json()            # 完整 JSON 导出
page.trace.clear()                # 清空缓冲区

# TraceEntry 属性
e.category / e.event / e.data / e.context_id / e.elapsed_ms / e.status
e.to_dict()

失败快照 API（异常自动抓现场）：

Settings.raise_when_ele_not_found = True
try:
    page.ele("#this-element-does-not-exist", timeout=2)
except ElementNotFoundError as e:
    snap = e.diagnostics            # FailureSnapshot
    snap.error_type / snap.error_message / snap.url / snap.context_id
    snap.screenshot_path / snap.dom_path / snap.recent_requests / snap.trace_entries
    snap.summary() / snap.to_json() / snap.to_dict()

敏感字段自动脱敏：password、cookie、authorization 等字段在诊断数据中自动替换为 ***，超长字符串自动截断。

33. JS 调试器​

参考示例 55_js_debugger.py、56_ai_autonomous_debug.py。ruyipage 通过 Firefox 自己的 DevTools 通道提供完整 JS 断点调试能力（断点、单步、调用栈、作用域、监视点、事件 / XHR 断点等）。

opts = FirefoxOptions()
opts.set_browser_path(BROWSER_PATH)
opts.enable_debugger()
page = FirefoxPage(opts)
page.get(PAGE)

dbg = page.debugger.start(auto_resume_after=90)

# 源码
dbg.sources(url_contains=SOURCE)
dbg.source_text(SOURCE)
dbg.breakable_lines(SOURCE)

# 断点 / 条件断点
bp = dbg.set_breakpoint(SOURCE, line, condition="i === 0")
dbg.breakpoints

# 等待暂停
state = dbg.wait_paused(timeout=30)
state.why / state.line
dbg.frames()                 # 调用栈
dbg.scope()                  # 作用域
dbg.step_into() / step_over() / step_out() / restart_frame()

# 日志断点（不暂停）
dbg.set_breakpoint(SOURCE, line, log_value="i, sum")
dbg.wait_logs(count=3, timeout=10)

# 暂停时检查对象
dbg.expand(probe, depth=1, max_items=20)
dbg.get_property(probe, "double")
dbg.entries(map_obj)
dbg.call(fn, args=[21])
dbg.invoke_getter(probe, "computed")
dbg.promise_state(promise)
dbg.global_object()

# 属性监视点
dbg.watch_property(config, "token", on="set")
dbg.unwatch_property(config)

# 异常暂停 / debugger 语句
dbg.pause_on_exceptions(True, ignore_caught=True)
dbg.pause_on_debugger_statement(True)

# 事件 / XHR 断点
dbg.available_event_breakpoints()
dbg.set_event_breakpoints(["event.mouse.click"])
dbg.set_xhr_breakpoint("/", "ANY")

# 控制
dbg.skip_breakpoints(True)   # 跳过所有断点
dbg.blackbox(SOURCE)         # 黑盒（不进入）
dbg.include_async_frames(True)
dbg.clear_breakpoints()
dbg.stop()

重要：JS 暂停期间，依赖它的 BiDi 调用（run_js、点击、取文本）会阻塞到超时。触发断点的调用必须放到后台线程里执行（参考示例 55 的 bg() 辅助函数）。

常见要点提醒​

  1. 用完关闭：每个脚本 finally 里调用 page.quit()，避免残留 Firefox 进程。
  2. 网络拦截必须放行：beforeRequestSent 阶段不处理的请求要显式 req.continue_request()，否则会挂起。
  3. Mock 跨域要带 CORS 头：mock 响应时加上 access-control-allow-origin，否则浏览器拦截。
  4. 原生优先、兼容保底：下拉选择 native_only 失败后回退 compat。
  5. JS 断点用后台线程：暂停期间 BiDi 调用会阻塞。
  6. 抓包先 start 再开页：page.capture / listen 无法补抓历史请求。
  7. 拟人动作 = 更真实：需要 isTrusted=true 的物理事件，用 page.actions 而非 JS 触发。
